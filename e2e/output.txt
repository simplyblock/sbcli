Run NAMESPACE=simplyblock
  NAMESPACE=simplyblock
  IFC_NAMES="br-ex:ens16np0"
  MGMT_IFC="${IFC_NAMES%%:*}"
  DATA_NICS="${IFC_NAMES#*:}"
  MAX_SUBSYS="30"
  SB_REPO="docker.io/simplyblock/simplyblock"
  SB_TAG="main"
  SPDK_IMAGE="docker.io/simplyblock/spdk:main-latest"
  # spdkImage and spdkProxyImage were StorageNodeSet fields. In v1alpha2
  # they survive only as a per-StorageNode override
  # (StorageNode.spec.config.spdkImage), and a ClusterDeploymentConfig
  # has no field that sets them -- the nodes it creates leave the image
  # empty and the control plane picks its own default.
  #
  # There is no non-racy way to pin it from here: the nodes are created
  # and added by the expansion, so anything this job patched afterwards
  # would land after the image had already been chosen. Rather than
  # patch and hope, the run says the pin did not happen, because an
  # lblk run silently testing the default SPDK build instead of the one
  # it named is a wrong green.
  # Exported so the bring-up checks it rather than assumes it: the
  # document cannot set spdkImage, so the run verifies what the nodes
  # actually came up on and fails if it is not what was asked for.
  export SPDK_IMAGE
  PARTITIONS="${PARTITIONS:-0}"
  JM_COUNT="${JM_COUNT:-3}"
  # 0 (the default) means derive vcpuCount from the node's CPU count below;
  # any other value in cluster_params is used verbatim.
  VCPU_COUNT_PARAM=$(echo "${VCPU_COUNT_PARAM:-0}" | tr -d '[:space:]')
  OPENSHIFT_CLUSTER="true"
  CORE_PERCENTAGE="50"
  FORCE_FORMAT_4K="true"
  DRIVE_SIZE_RANGE="1.7T-2T"
  CLUSTER_ENV="openshift-local"
  if [ "$CLUSTER_ENV" = "local" ]; then
    :
  fi
  
  # Convert partitions to enableJournalDevice (inverted: 0→true, 1→false)
  if [ "${PARTITIONS}" = "0" ]; then
    ENABLE_JOURNAL_DEVICE="true"
  else
    ENABLE_JOURNAL_DEVICE="false"
  fi
  
  # vcpuCount: an explicit vcpu_count in cluster_params wins; 0 means calculate it
  # from CORE_PERCENTAGE by querying the actual CPU count on a worker node.
  if [ "${VCPU_COUNT_PARAM}" -gt 0 ] 2>/dev/null; then
    VCPU_COUNT="${VCPU_COUNT_PARAM}"
    echo "Using vcpuCount=${VCPU_COUNT} from cluster_params (skipping CPU-count query)"
  else
    FIRST_WORKER="worker-1,worker-2,worker-3,worker-4"
    FIRST_WORKER="${FIRST_WORKER%%,*}"
    if [ "$OPENSHIFT_CLUSTER" = "true" ]; then
      TOTAL_CPUS=$(oc debug node/"${FIRST_WORKER}" -- chroot /host nproc 2>/dev/null || echo "0")
    else
      TOTAL_CPUS=$(kubectl get node "${FIRST_WORKER}" -o jsonpath='{.status.capacity.cpu}' 2>/dev/null || echo "0")
    fi
    TOTAL_CPUS=$(echo "${TOTAL_CPUS}" | tr -d '[:space:]')
    if [ "${TOTAL_CPUS}" -gt 0 ] 2>/dev/null; then
      VCPU_COUNT=$(( TOTAL_CPUS * CORE_PERCENTAGE / 100 ))
      [ "${VCPU_COUNT}" -lt 1 ] && VCPU_COUNT=1
    else
      echo "WARNING: Could not determine CPU count, defaulting vcpuCount to 6"
      VCPU_COUNT=6
    fi
    echo "Calculated vcpuCount=${VCPU_COUNT} (${TOTAL_CPUS} CPUs * ${CORE_PERCENTAGE}%)"
  fi
  
  RESERVED_SYSTEM_CPU=""
  if [ "openshift-local" = "openshift-baremetal" ]; then
    RESERVED_SYSTEM_CPU="0,1,10,11,16,17,26,27"
  fi
  
  PCIE_MODEL=""
  CLUSTER_ENV="openshift-local"
  if [ "$CLUSTER_ENV" != "openshift-baremetal" ] && [ "$CLUSTER_ENV" != "openshift-local" ]; then
    PCIE_MODEL="SAMSUNG MZQLB1T9HAJR-00007"
  fi
  
  # Which distribution the operator is deploying onto. It is a
  # shorthand: naming OpenShift once sets openShiftCluster,
  # enableCpuTopology and enableKubeletConfiguration together, so the
  # three flags this workflow used to compute individually are now one
  # word (expansion.go buildWorkload).
  case "$CLUSTER_ENV" in
    aws-openshift|openshift-local|openshift-baremetal) ENVIRONMENT="OpenShift" ;;
    talos)                                             ENVIRONMENT="Talos" ;;
    *)                                                 ENVIRONMENT="Vanilla" ;;
  esac
  
  # nvme unless cluster_params says otherwise. Defaulted where it
  
  # is used, so it holds however the value arrived.
  
  DEVICE_MODE="${DEVICE_MODE:-nvme}"
  
  #
  
  # lblk selectors and the NVMe PCIe selectors are mutually
  
  # exclusive under the StorageNodeSet CEL rule, which tests
  
  # size(...) > 0 -- so blanking is enough, the keys may stay.
  
  if [ "${DEVICE_MODE}" = "lblk" ]; then
    # Mandatory on lblk: add-node refuses with "lblk device mode
    # requires --enable-journal-device; partitioned journal mode is not
    # supported".
    ENABLE_JOURNAL_DEVICE="true"
    # Both are NVMe selectors, and discovery scans one class per run.
    # Sending them with enableLogicalBlockDevices selects nothing.
    DRIVE_SIZE_RANGE=""
    PCIE_MODEL=""
    echo "lblk mode: block-device discovery, journal device on, PCIe selectors blanked"
  fi
  
  # Conditionally build backup spec block
  BACKUP_SPEC=""
  CLUSTER_SEC="none"
  TESTNAME_INPUT="LblkOutageMatrixK8s"
  TESTNAME_LOWER="$(echo "$TESTNAME_INPUT" | tr '[:upper:]' '[:lower:]')"
  if [ "$CLUSTER_SEC" = "backup" ] || echo "$TESTNAME_LOWER" | grep -q "backup"; then
    BACKUP_SPEC="  backup:
      credentialsSecretRef:
        name: backup-credentials
      localEndpoint: ${MINIO_ENDPOINT}
      localTesting: true
      secondaryTarget: 0
      snapshotBackups: true
      withCompression: false"
  fi
  
  # Build vault settings and encryption pool blocks (conditional on TLS)
  # Always configured, for the same reason as TLS_FLAGS above.
  VAULT_SETTINGS=""
  ENCRYPTION_POOL_YAML=""
  if true; then
    VAULT_SETTINGS="  hashicorpVaultSettings:
      baseURL: \"https://openbao.vault:8200\""
    ENCRYPTION_POOL_YAML="---
  apiVersion: storage.simplyblock.io/v1alpha1
  kind: StoragePool
  metadata:
    name: encryption-pool
    namespace: ${NAMESPACE}
  spec:
    clusterName: simplyblock-cluster
    storageClassParameters:
      encryption: true"
  fi
  
  echo "=== Bringing the cluster up through the deployment config ==="
  # StorageNodeSet is not reconciled any more. operator/cmd/main.go
  # registers no reconciler for it and deployment/expansion.go is the
  # only thing that creates a StorageNode, so applying a set succeeded
  # and produced nothing -- the cluster then waited forever for storage
  # nodes nobody had asked for.
  #
  # The supported path is a ClusterDeploymentConfig: discovery inspects
  # the workers and writes a draft, we write our parameters into it, and
  # approving it builds the StorageCluster and its StorageNodes. The
  # draft is immutable once approved, so everything has to be set first;
  # the script does that in order.
  export NAMESPACE
  export CLUSTER_NAME="simplyblock-cluster"
  export DEVICE_MODE NDCS NPCS VCPU_COUNT MAX_SUBSYS
  export MGMT_IFC DATA_NICS
  export JM_COUNT
  export JM_PERCENT="3"
  export ENABLE_JOURNAL_DEVICE
  export ENVIRONMENT
  export DRIVE_SIZE_RANGE PCIE_MODEL
  # No default in the CRD and immutable on the cluster once set,
  # so an unstated fabric is not a thing that can be fixed later.
  export FABRIC_TYPE="tcp"
  # One field, two operations: the operator resolves it to a 4K reformat
  # on an NVMe cluster and a wipefs on a block one.
  export FORCE_DRIVE_FORMAT="${FORCE_FORMAT_4K}"
  export WORKER_NODES="worker-1,worker-2,worker-3,worker-4"
  # On lblk the workers boot from /dev/sda; without this the draft
  # offers the root disk as storage and the approval gate is the only
  # thing standing between that and a reformat.
  if [ "${DEVICE_MODE}" = "lblk" ]; then
    export BLOCK_DENY_LIST="${LBLK_DENY_LIST:-/dev/sda}"
  fi
  
  python3 $GITHUB_WORKSPACE/e2e/scripts/k8s_cluster_bringup.py
  
  echo "=== Settings the deployment config does not carry ==="
  # A deployment config describes a deployment, not every knob of a
  # cluster: thresholds, node affinity, backup and vault have no field
  # on it. They are mutable on the StorageCluster, so they are patched
  # onto the cluster the expansion just created.
  #
  # Patched through v1alpha1 deliberately. That version is still served
  # and carries these in exactly the shapes this workflow already
  # builds -- backup's localEndpoint/localTesting/secondaryTarget among
  # them -- and the conversion webhook translates. Rewriting them into
  # v1alpha2's own backup shape by hand would be a silent guess, and a
  # wrong guess reads as a backup test failing for its own reasons.
  cat > /tmp/cluster-extra.yaml <<EOF
  spec:
    enableNodeAffinity: true
    warningThreshold:
      capacity: 95
      provisionedCapacity: 97
    criticalThreshold:
      capacity: 96
      provisionedCapacity: 98
  ${BACKUP_SPEC}
  ${VAULT_SETTINGS}
  EOF
  echo "--- cluster patch ---"; cat /tmp/cluster-extra.yaml
  kubectl -n ${NAMESPACE} patch \
    storageclusters.v1alpha1.storage.simplyblock.io simplyblock-cluster \
    --type=merge --patch-file /tmp/cluster-extra.yaml
  
  # reservedSystemCPU was a StorageNodeSet field and is now part of the
  # cluster's storage-node workload, so it has no v1alpha1 home and is
  # patched on v1alpha2 directly.
  if [ -n "${RESERVED_SYSTEM_CPU}" ]; then
    kubectl -n ${NAMESPACE} patch storagecluster simplyblock-cluster \
      --type=merge \
      -p "{\"spec\":{\"storageNodes\":{\"reservedSystemCPU\":\"${RESERVED_SYSTEM_CPU}\"}}}"
  fi
  
  echo "=== Applying pools ==="
  cat <<EOF | kubectl apply -f -
  apiVersion: storage.simplyblock.io/v1alpha1
  kind: StoragePool
  metadata:
    name: simplyblock-pool
    namespace: ${NAMESPACE}
  spec:
    clusterName: simplyblock-cluster
  ${ENCRYPTION_POOL_YAML}
  EOF
  
  echo "Custom resources applied"
  sleep 10
  shell: /usr/bin/bash -e {0}
  env:
    TESTNAME: LblkOutageMatrixK8s
    KEY_PATH: /home/ec2-user/.ssh/simplyblock-us-east-2.pem
    CLIENT_IPS: 
    NFS_MOUNTPOINT: /mnt/nfs_share/
    SSH_USER: root
    SSH_PASSWORD: ***
    KEY_NAME: simplyblock-us-east-2.pem
    KUBECONFIG: /home/ec2-user/.kube/config-36106621262
    NDCS: 1
    NPCS: 1
    BS: 4096
    CHUNK_BS: 4096
    PARTITIONS: 1
    JM_COUNT: 4
    VCPU_COUNT_PARAM: 0
    DEVICE_MODE: lblk
    FLUENTBIT_PATCHED: true
